# Dawa24 — public marketplace pages are indexable; authenticated areas are not.
#
# robots.txt is a request, not a control. Everything below is also enforced in
# code (internal/platform/antiscrape); this file exists so that well-behaved
# crawlers never trip those defences, and so that a crawler which ignores it has
# demonstrably chosen to.

Disallow: /admin/
Disallow: /vendor/
Disallow: /account/
Disallow: /customer/

# Faceted URLs multiply one listing into thousands of near-identical pages.
# Crawling them wastes budget on our side and looks exactly like harvesting.
Disallow: /*?*page=
Disallow: /*?*page_size=
Disallow: /*?*sort=
Disallow: /*?*min_price=
Disallow: /*?*max_price=
Disallow: /*?*in_stock=
Disallow: /*?*has_discount=

Crawl-delay: 5
