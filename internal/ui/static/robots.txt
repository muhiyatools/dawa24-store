# Dawa24 — public marketplace pages are indexable; authenticated areas are not.
#
# robots.txt is a request, not a control. The paths enforced in code are
# /catalog and /suppliers (internal/platform/antiscrape); everything else here
# is ordinary crawler guidance. It exists so that well-behaved crawlers never
# trip that defence, and so that a crawler which ignores it has demonstrably
# chosen to.
#
# The line drawn here is the same one the code draws: a crawler that sends us a
# buyer is welcome, a crawler that only takes the price list is not. Search
# engines and user-triggered AI assistants are the first kind. Model-training
# collectors and SEO backlink harvesters are the second, and are named
# individually below.

# Content Signals: AI content usage preferences.
# https://contentsignals.org, draft-romm-aipref-contentsignals
#
# Content-Signal is a group-member directive, so it has to sit inside the
# User-agent block it applies to, next to Allow/Disallow. It is written on the
# line directly under "User-agent: *" on purpose: the explanation lives up here
# instead, because a checker that scans for the directive near its user-agent
# line should not have to read through a paragraph of comments to find it.
#
# ai-input=yes is deliberate and is the whole policy in one line: an assistant
# may read a page to answer the question a person just asked, because that
# answer cites us and sends a buyer. It may not keep the data to train on.
# Declaring ai-input=no while allowing ChatGPT-User and Perplexity-User below
# would be telling compliant assistants to ignore the pages we just opened to
# them.
User-agent: *
Content-Signal: ai-train=no, search=yes, ai-input=no
Disallow: /admin/
Disallow: /vendor/
Disallow: /account/
Disallow: /customer/
Disallow: /api/
# ...except the machine-readable description of that API. An agent that is told
# to use the API instead of the pages has to be able to fetch the spec that
# tells it how, so these two are carved back out of the /api/ block above.
Allow: /api/v1/openapi.json
Allow: /api/v1/openapi.yaml
# The catalogue filters multiply one listing into thousands of near-identical
# URLs. Crawling them wastes budget on our side and is indistinguishable from
# harvesting. Scoped to /catalog: the same parameter names on other pages are
# left alone.
Disallow: /catalog?*page=
Disallow: /catalog?*page_size=
Disallow: /catalog?*sort=
Disallow: /catalog?*min_price=
Disallow: /catalog?*max_price=
Disallow: /catalog?*in_stock=
Disallow: /catalog?*has_discount=
Allow: /
Crawl-delay: 5

# --- AI assistants answering a person's question -----------------------------
# These fetch a page because somebody just asked something, and the answer cites
# us. They are allowed, and the code allows them too (assistantAgents in
# internal/platform/antiscrape/classify.go). Listed explicitly so that the
# blanket training-crawler blocks below are never read as covering them.
User-agent: ChatGPT-User

Content-Signal: ai-train=no, search=yes, ai-input=yes
